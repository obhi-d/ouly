Unit Test Examples

The excerpts below are included straight from the test sources, so they always show the code that CI compiles and runs.

Scheduler: continuations

then attaches continuations that the scheduler queues once the producer finishes.

TEST_CASE("scheduler tasks chain queued continuations", "[scheduler][task][then]")
{
  ouly::scheduler scheduler;
  scheduler.create_group(ouly::workgroup_id(0), 0, 4);
  scheduler.begin_execution();
  auto const& ctx = ouly::task_context::this_context::get();

  auto producer = ouly::submit_task(ctx,
                                    []() -> int
                                    {
                                      return 21;
                                    });
  auto doubled  = producer.then(ctx,
                                [](int value) -> int
                                {
                                 return value * 2;
                               });
  auto text     = producer.then(ctx,
                                [](int value) -> std::size_t
                                {
                              return static_cast<std::size_t>(value);
                            });

  REQUIRE(doubled.get(ctx) == 42);
  REQUIRE(text.get(ctx) == 21);
  scheduler.end_execution();
}

Scheduler: scoped joins

task_scope::join(ctx) runs only the scope’s own children while it waits.

TEST_CASE("task_scope cooperative join executes only its own children", "[scheduler][task][scope]")
{
  ouly::scheduler scheduler;
  scheduler.create_group(ouly::workgroup_id(0), 0, 1);
  scheduler.begin_execution();
  auto const& ctx = ouly::task_context::this_context::get();

  std::atomic_bool unrelated{false};
  std::atomic_bool child{false};
  scheduler.submit(ctx,
                   [&unrelated](ouly::task_context const&) noexcept
                   {
                     unrelated.store(true, std::memory_order_release);
                   });

  ouly::task_scope scope;
  scope.run(ctx,
            [&child]() noexcept
            {
              child.store(true, std::memory_order_release);
            });
  scope.join(ctx);

  REQUIRE(child.load(std::memory_order_acquire));
  REQUIRE_FALSE(unrelated.load(std::memory_order_acquire));
  scheduler.wait_for_tasks();
  REQUIRE(unrelated.load(std::memory_order_acquire));
  scheduler.end_execution();
}

auto_parallel_for: range splitting

The range pool behind auto_parallel_for splits a range into disjoint pieces that cover it exactly once.

TEST_CASE("range_pool: split_to_fill partitions the range without overlap", "[auto_parallel_for][range_pool]")
{
  using range_t = ouly::subrange<uint32_t>;
  ouly::range_pool<range_t, 8> pool(range_t{0, 1000});
  pool.split_to_fill(3, 1);
  REQUIRE(pool.size() > 1);

  std::vector<uint32_t> hits(1000, 0);
  while (!pool.empty())
  {
    auto r = pool.front();
    REQUIRE(pool.front_depth() <= 3);
    for (auto i = r.begin(); i != r.end(); ++i)
    {
      ++hits[i];
    }
    pool.pop_front();
  }
  for (auto h : hits)
  {
    REQUIRE(h == 1);
  }

  // a range that cannot be split stays as is
  ouly::range_pool<range_t, 8> single(range_t{0, 1});
  single.split_to_fill(3, 1);
  REQUIRE(single.size() == 1);
}

small_vector: storage mode

Once on the heap, a small_vector keeps its buffer until shrink_to_fit.

TEST_CASE("small_vector: heap buffer is kept until shrink_to_fit", "[small_vector][capacity]")
{
  ouly::small_vector<int, 4> v;
  REQUIRE(v.is_inlined());

  // reserve works even while the contents would fit inline
  v.reserve(100);
  REQUIRE(!v.is_inlined());
  REQUIRE(v.capacity() >= 100);
  auto const* buffer = v.data();

  for (int round = 0; round < 3; ++round)
  {
    for (int i = 0; i < 64; ++i)
    {
      v.push_back(i);
    }
    REQUIRE(v.data() == buffer); // no reallocation, clear() kept the buffer
    v.clear();
    REQUIRE(v.empty());
    REQUIRE(v.capacity() >= 100);
  }

  // crossing the inline threshold with push/pop does not bounce between storages
  for (int i = 0; i < 5; ++i)
  {
    v.push_back(i);
  }
  v.pop_back();
  REQUIRE(!v.is_inlined());
  REQUIRE(v.data() == buffer);
  v.push_back(4);
  REQUIRE(v.data() == buffer);

  v.resize(2);
  REQUIRE(!v.is_inlined());
  v.shrink_to_fit();
  REQUIRE(v.is_inlined());
  REQUIRE(v.size() == 2);
  REQUIRE(v[0] == 0);
  REQUIRE(v[1] == 1);

  // moving a heap vector hands over the buffer
  v.reserve(50);
  v.push_back(7);
  auto const*                moved_buffer = v.data();
  ouly::small_vector<int, 4> w(std::move(v));
  REQUIRE(w.data() == moved_buffer);
  REQUIRE(w.size() == 3);
  REQUIRE(v.empty());
  REQUIRE(v.is_inlined());

  ouly::small_vector<int, 4> x;
  x.reserve(20);
  x = std::move(w); // x's own buffer is released, w's taken over (checked by LeakSanitizer)
  REQUIRE(x.data() == moved_buffer);
  REQUIRE(x[2] == 7);

  // swapping heap and inline vectors
  ouly::small_vector<int, 4> y{1, 2};
  x.swap(y);
  REQUIRE(y.data() == moved_buffer);
  REQUIRE(x.size() == 2);
  REQUIRE(x.is_inlined());
  REQUIRE(x[1] == 2);
}

coalescing_arena_allocator: retained arenas

One emptied arena per memory class is kept for reuse until trim.

TEST_CASE("coalescing_arena_allocator: emptied arenas are retained for reuse until trimmed",
          "[coalescing_arena_allocator][retain]")
{
  arena_memory                     memory;
  ouly::coalescing_arena_allocator allocator{1024};
  REQUIRE(allocator.get_retained_empty_arenas() == 1);

  // a per-frame alloc/free of one object must not create and destroy an arena every frame
  for (int frame = 0; frame < 8; ++frame)
  {
    auto a = allocator.allocate(128, memory);
    REQUIRE(memory.active_ == 1);
    allocator.deallocate(a.get_allocation_id(), memory);
    REQUIRE(memory.active_ == 1);
    allocator.validate_integrity();
  }

  // only one empty arena per class is kept; a second emptied arena of the class is released
  auto first  = allocator.allocate(1000, memory);
  auto second = allocator.allocate(1000, memory);
  REQUIRE(memory.active_ == 2);
  allocator.deallocate(first.get_allocation_id(), memory);
  allocator.deallocate(second.get_allocation_id(), memory);
  REQUIRE(memory.active_ == 1);

  // other memory classes keep their own empty arena
  auto other = allocator.allocate(64, memory, ouly::ca_allocation_options{.memory_class_ = 3});
  allocator.deallocate(other.get_allocation_id(), memory);
  REQUIRE(memory.active_ == 2);

  // dedicated arenas are always released
  auto dedicated = allocator.allocate(4096, memory);
  REQUIRE(memory.active_ == 3);
  allocator.deallocate(dedicated.get_allocation_id(), memory);
  REQUIRE(memory.active_ == 2);
  allocator.validate_integrity();

  REQUIRE(allocator.trim(memory) == 2);
  REQUIRE(memory.active_ == 0);
  REQUIRE(allocator.trim(memory) == 0);

  // retaining nothing restores immediate release
  allocator.set_retained_empty_arenas(0);
  auto again = allocator.allocate(128, memory);
  allocator.deallocate(again.get_allocation_id(), memory);
  REQUIRE(memory.active_ == 0);
  allocator.validate_integrity();
}

Memory-mapped file ranges

map(path, offset, length) exposes length bytes starting at offset.

TEST_CASE("Validate mmap_source with offset", "[mmap_source]")
{
  const std::string filename = "test_mmap_offset.dat";
  const std::string content  = "0123456789abcdefghij";
  {
    std::ofstream file(filename, std::ios::binary);
    file.write(content.c_str(), static_cast<std::streamsize>(content.size()));
  }

  {
    ouly::mmap_source source;
    source.map(filename, 5, 10);
    REQUIRE(source.size() == 10);
    REQUIRE(static_cast<char>(source[0]) == '5');
    REQUIRE(static_cast<char>(source[9]) == 'e');

    auto tail = ouly::make_mmap_source(filename, 15, ouly::map_entire_file);
    REQUIRE(tail.size() == 5);
    REQUIRE(static_cast<char>(tail[0]) == 'f');
  }

  cleanup_test_file(filename);
}

ECS: collections

TEMPLATE_TEST_CASE("collection: for_each visits exactly the inserted set across word and pool boundaries",
                   "[ecs][collection]", ouly::ecs::entity<>, ouly::ecs::rxentity<>)
{
  using E = TestType;
  ouly::ecs::collection<E>      col;
  ouly::ecs::components<int, E> comp;
  std::set<uint32_t> const      ids = {0, 1, 63, 64, 65, 127, 128, 200, 4095, 4096, 5000};

  for (auto id : ids)
  {
    col.emplace(E(id));
    comp.emplace_at(E(id), static_cast<int>(id));
  }

  std::set<uint32_t> visited;
  col.for_each(comp,
               [&](E e, int& v)
               {
                 REQUIRE(static_cast<uint32_t>(v) == e.get());
                 visited.insert(e.get());
               });
  REQUIRE(visited == ids);

  // Ranged iteration honors [first, last)
  std::set<uint32_t> ranged;
  col.for_each(comp, 63, 129,
               [&](E e, int&)
               {
                 ranged.insert(e.get());
               });
  REQUIRE(ranged == std::set<uint32_t>{63, 64, 65, 127, 128});

  // Erased entities disappear from iteration
  col.erase(E(64));
  col.erase(E(4096));
  std::set<uint32_t> after_erase;
  col.for_each(comp,
               [&](E e, int&)
               {
                 after_erase.insert(e.get());
               });
  std::set<uint32_t> expected = ids;
  expected.erase(64);
  expected.erase(4096);
  REQUIRE(after_erase == expected);

  // Const overload matches
  auto const&        ccol = col;
  std::set<uint32_t> cvisited;
  ccol.for_each(static_cast<ouly::ecs::components<int, E> const&>(comp),
                [&](E e, int const&)
                {
                  cvisited.insert(e.get());
                });
  REQUIRE(cvisited == expected);
}

microexpr: operators and identifiers

TEST_CASE("microexpr: logical operators, negation and identifiers", "[microexpr]")
{
  std::unordered_map<std::string_view, int> table = {
   {      "A", 1},
   {      "B", 0},
   {"MY_FLAG", 3},
  };

  ouly::microexpr expr(
   [&table](std::string_view v) -> std::optional<int>
   {
     auto it = table.find(v);
     if (it != table.end())
       return {it->second};
     return {};
   });

  // the right operand of && / || must be consumed even when the result is already known
  REQUIRE(expr.evaluate("0 && 1 ? 5 : 0") == false);
  REQUIRE(expr.evaluate("0 && 1 ? 0 : 5") == true);
  REQUIRE(expr.evaluate("(1 || B)") == true);
  REQUIRE(expr.evaluate("(A || B)") == true);
  REQUIRE(expr.evaluate("(B && A) == 0") == true);

  REQUIRE(expr.evaluate("!B") == true);
  REQUIRE(expr.evaluate("!A") == false);
  REQUIRE(expr.evaluate("!$UNDEFINED") == true);
  REQUIRE(expr.evaluate("$(A) && !$(UNDEFINED)") == true);
  REQUIRE(expr.evaluate("$( A ) && $(B)") == true);

  REQUIRE(expr.evaluate("MY_FLAG == 3") == true);
  REQUIRE(expr.evaluate("$MY_FLAG") == true);

  // malformed input must not read past the end
  REQUIRE(expr.evaluate("(1") == false);
  REQUIRE(expr.evaluate("$(A") == false);
  REQUIRE(expr.evaluate("") == false);
}
